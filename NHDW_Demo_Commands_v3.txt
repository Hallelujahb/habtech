# nhdw demo — real facility data, commands top to bottom
#
# this is the whole pipeline, nothing outside it. everything lives at
# ~/habtech. run habtech_setup.sh once first, then this file, in order.
# credentials for everything: username/password, postgres user is
# "postgres" (see CREDENTIALS.md).
#
# steps 1-9 are the original ingest -> dbt -> spark -> minio -> clickhouse
# -> LLM-linking -> monitoring pipeline. steps 10-13 are the newer
# additions (dbt Python models, Kafka/Debezium CDC, Superset, Keycloak
# across all of it) -- each of those points at its own longer doc rather
# than being fully inlined here, since each is a multi-container setup of
# its own. step 14 is the infra-as-code layer (Helm/Kubernetes, Ansible,
# Airflow, GitHub Actions) -- validated/lint-passing artifacts, not a
# live deployment; see NHDW_Complete_System_Overview.pdf section 4 and
# DEMO.md for how to present that distinction honestly.
#
# the datasets, all real, all explained in DATASETS_EXPLAINED.txt:
#   - master_facility_registry.csv   1,332 rows  (Master Facility Registry)
#   - org_units.csv                    492 rows  (DHIS2 org unit hierarchy)
#   - facility_linkage_unresolved.csv  152 rows  (org units with no MFR match)

#                                                  used for LLM name matching)
#
# copy the habtech folder to ~/habtech before starting.

export HABTECH_HOME=~/habtech

# ---------------------------------------------------------------------
# step 1, containers — everything the pipeline sits on top of
# ---------------------------------------------------------------------
# every container below has --restart unless-stopped, so once it's up
# once, it survives a reboot on its own -- Docker's daemon restarts it
# automatically, as long as the Docker daemon itself starts on boot
# (habtech_setup.sh runs `systemctl enable docker` for exactly this).
#
# IMPORTANT, the actual boundary: --restart makes a STOPPED container
# come back, with its data intact. it does NOT protect against re-running
# this block, which starts with `docker rm -f` -- that deletes the
# containers outright, wiping whatever was inside (ingested tables, dbt
# models, everything from steps 2-9). rule of thumb: after the first
# real setup, don't rerun this block again unless you actually want a
# clean slate. if a container just stopped (crashed, machine slept,
# `docker logs <name>` shows nothing alarming), `docker start <name>`
# brings it back with data intact -- that's the normal recovery path,
# not rerunning step 1.

sudo docker rm -f pg clickhouse minio keycloak prometheus grafana cadvisor 2>/dev/null

sudo docker run --name pg --restart unless-stopped -e POSTGRES_PASSWORD=password -p 5432:5432 -d postgres:16
sudo docker run -d --name clickhouse --restart unless-stopped -p 8123:8123 -p 9363:9363 \
  -e CLICKHOUSE_USER=username -e CLICKHOUSE_PASSWORD=password -e CLICKHOUSE_DEFAULT_ACCESS_MANAGEMENT=1 \
  -v ~/habtech/monitoring/clickhouse_prometheus.xml:/etc/clickhouse-server/config.d/prometheus.xml \
  clickhouse/clickhouse-server
sudo docker run -d --name minio --restart unless-stopped -p 9000:9000 -p 9001:9001 -e "MINIO_ROOT_USER=username" -e "MINIO_ROOT_PASSWORD=password" -e "MINIO_PROMETHEUS_AUTH_TYPE=public" minio/minio server /data --console-address ":9001"
sleep 8

sudo docker run -d --name keycloak --restart unless-stopped -p 8180:8080 -p 9002:9000 --add-host=host.docker.internal:host-gateway -e KEYCLOAK_ADMIN=username -e KEYCLOAK_ADMIN_PASSWORD=password quay.io/keycloak/keycloak:latest start-dev --metrics-enabled=true
sleep 45
# login is username / password. this brings Keycloak up standalone --
# wiring it into Grafana, MinIO, and gateways in front of Postgres/
# ClickHouse's web UIs is its own step, see KEYCLOAK_SSO_SETUP.md
# (needs Keycloak up first, run it any time after this point)

sudo docker run \
  --volume=/:/rootfs:ro --volume=/var/run:/var/run:ro --volume=/sys:/sys:ro \
  --volume=/var/lib/docker/:/var/lib/docker:ro --volume=/dev/disk/:/dev/disk:ro \
  --publish=8085:8080 --detach=true --restart unless-stopped --name=cadvisor gcr.io/cadvisor/cadvisor:latest

sudo docker run -d --name prometheus --restart unless-stopped -p 9090:9090 \
  --add-host=host.docker.internal:host-gateway \
  -v ~/habtech/monitoring/prometheus.yml:/etc/prometheus/prometheus.yml \
  prom/prometheus
sleep 5
# confirm two targets at http://localhost:9090/targets, both green,
# before moving on — that's the actual proof monitoring is watching

sudo docker run -d --name grafana --restart unless-stopped -p 3000:3000 \
  --add-host=host.docker.internal:host-gateway \
  -e GF_SECURITY_ADMIN_USER=username -e GF_SECURITY_ADMIN_PASSWORD=password \
  grafana/grafana
sleep 15
# http://localhost:3000, login username / password

sudo docker exec -it minio mc alias set localminio http://localhost:9000 username password 2>/dev/null || true
python3 ~/habtech/lakehouse/setup_minio_buckets.py

# ---------------------------------------------------------------------
# step 2, data cleaning — real transformations, run BEFORE ingestion
# ---------------------------------------------------------------------
# cleaning has to happen before the join, not after: a join is only as
# good as the keys and names going into it. this script trims whitespace,
# fixes lowercase-leading facility names, normalizes zone naming so it can
# be matched by name (org_units says "Fafan Zone", mfr says "Fafan"),
# validates coordinates against Ethiopia's real bounding box, converts
# year_opened from the Ethiopian calendar to a Gregorian year, and flags
# which beds/rooms fields were actually reported vs silently blank.
# every number below is real, printed by the script itself.

cd ~/habtech/data_cleaning
python3 clean_facility_data.py
cat ../DATA_CLEANING_LOG.md

# ---------------------------------------------------------------------
# step 3, meltano ingestion — the cleaned files
# ---------------------------------------------------------------------

cd ~/habtech/meltano_project
meltano config set tap-csv files '[
  {"entity": "mfr_facilities", "path": "'"$HOME"'/habtech/data_v2/master_facility_registry_clean.csv", "keys": ["id"]},
  {"entity": "org_units", "path": "'"$HOME"'/habtech/data_v2/org_units_clean.csv", "keys": ["id"]},
  {"entity": "facility_linkage_unresolved", "path": "'"$HOME"'/habtech/data_v2/facility_linkage_unresolved.csv", "keys": ["id"]}
]'

meltano config set target-postgres host localhost
meltano config set target-postgres port 5432
meltano config set target-postgres user postgres
meltano config set target-postgres password password
meltano config set target-postgres database postgres
meltano config set target-postgres default_target_schema public

time meltano run tap-csv target-postgres

# confirm the three real tables landed
sudo docker exec -it pg psql -U postgres -c "\dt"
sudo docker exec -it pg psql -U postgres -c "SELECT count(*) FROM mfr_facilities;"

# ---------------------------------------------------------------------
# step 7, clickhouse — load the mfr rollup for fast Superset queries
# ---------------------------------------------------------------------

sudo docker cp ~/habtech/data_v2/master_facility_registry_clean.csv clickhouse:/tmp/mfr.csv
sudo docker exec -it clickhouse clickhouse-client --user username --password password --query "
CREATE TABLE IF NOT EXISTS default.mfr_facilities_ch (
    id UInt64, name String, type String, subtype String, ownership String,
    status String, operational_status String, region String, zone String,
    woreda String, city String, kebele String, latitude Nullable(Float64),
    longitude Nullable(Float64), altitude Nullable(Float64), dhis2_id String,

    number_of_icu_rooms UInt32, year_opened String, zone_normalized String,
    year_opened_gregorian Nullable(UInt16), beds_reported UInt8

sudo docker exec -i clickhouse clickhouse-client --user username --password password --query "
INSERT INTO default.mfr_facilities_ch FORMAT CSVWithNames" < ~/habtech/data_v2/master_facility_registry_clean.csv

cd ~/habtech/dbt_clickhouse

# ---------------------------------------------------------------------
# step 8, LLM-assisted linking — the part a plain join cannot do
# ---------------------------------------------------------------------
# the deterministic join in step 3/4 resolves ~69% of org units (340/492).
# the rest have no shared id, only a name. this step asks an LLM to judge
# name matches against the manually maintained regional registry. only a
# facility name and region ever get sent anywhere — see
# DATA_GOVERNANCE_NOTE.md for exactly what does and doesn't leave this
# machine and why that's the case.
#
# two options, same script logic, same prompt, pick one:

# option A: Gemini, hosted, needs an API key, generally the more reliable
# judgment call on borderline name matches
export GEMINI_API_KEY=your_key_here
cd ~/habtech/llm_examples
python3 llm_facility_matching.py

# option B: Ollama, fully local, no API key, nothing leaves this machine
# at all -- github.com/ollama/ollama, open source. noticeably less
# reliable on borderline cases with a small model, that tradeoff is real
# and worth saying out loud if this is the one used in the room
curl -fsSL https://ollama.com/install.sh | sh
ollama pull llama3.2
cd ~/habtech/llm_examples
python3 llm_facility_matching_local.py

# option C, the actual recommended one: LinkTransformer -- a record-linkage
# library, not a general chat LLM. local, no API key, no rate limit, gives
# a numeric similarity score instead of a free-text verdict to parse. see
# LINKING_LIBRARY_COMPARISON.md for why this is the one to lead with in
# the room over A/B above.
cd ~/habtech/linktransformer_examples
python3 linktransformer_facility_matching.py
python3 linktransformer_dedup_registry.py

# ---------------------------------------------------------------------
# step 10, dbt Python models -- the linkage scoring step as a real dbt
# model, not just a standalone script
# ---------------------------------------------------------------------
# see DBT_PYTHON_MODELS_NOTE.md first -- this only works on Postgres
# (via the dbt-postgres-python/fal adapter), NOT on ClickHouse, which has
# no dbt Python model support at all. the ClickHouse equivalent is a
# plain script, run separately, step 10b below.
#
# optional step -- only run this if habtech_setup.sh reported
# "dbt-postgres-python installed OK" (needs python3.10/3.11, a separate

# reported the package unavailable, skip step 10 entirely; steps 1-9
# are unaffected either way.
#
# TWO commands, TWO different venvs, in this order -- not one dbt run:
#   1. export_scored_matches.py runs LinkTransformer (needs pandas>=2.0)
#      and writes its output to a CSV.
#   2. dbt run reads that CSV (needs pandas<2.0, dbt_python_venv).
# these can't be merged into one step: dbt-postgres-python requires
# pandas<2.0 and LinkTransformer requires pandas>=2.0 -- genuinely
# incompatible in one Python process, on any Python version. see the
# comment at the top of dbt_postgres_python/models/fact_linkage_scored.py
# for the full explanation.

cd ~/habtech/linktransformer_examples
python3 export_scored_matches.py

cd ~/habtech/dbt_postgres_python
~/habtech/dbt_python_venv/bin/dbt run --profiles-dir profiles
sudo docker exec -it pg psql -U postgres -c "SELECT * FROM fact_linkage_scored ORDER BY similarity_score DESC LIMIT 10;"
